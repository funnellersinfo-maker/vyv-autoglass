    66→function formatPhone(v: string) {
    67→  const digits = v.replace(/\D/g, "").slice(0, 10);
    68→  if (digits.length === 0) return "";
    69→  if (digits.length < 4) return `(${digits}`;
    70→  if (digits.length < 7) return `(${digits.slice(0, 3)}) ${digits.slice(3)}`;
    71→  return `(${digits.slice(0, 3)}) ${digits.slice(3, 6)}-${digits.slice(6)}`;
    72→}

    74→function nextDates(count: number) {
    75→  const out: { iso: string; weekday: string; day: string; month: string }[] = [];
    76→  const today = new Date();
    77→  for (let i = 0; i < count; i++) {
    78→    const d = new Date(today);
    79→    d.setDate(today.getDate() + i);
    80→    out.push({
    81→      iso: d.toISOString().slice(0, 10),
    82→      weekday: d.toLocaleDateString("es-US", { weekday: "short" }),
    83→      day: String(d.getDate()),
    84→      month: d.toLocaleDateString("es-US", { month: "short" }),
    85→    });
    86→  }
    87→  return out;
    88→}
